Mobile Developer Implementation Guide
ð± Implementation Requirements for Mobile Developer
  1. Store Token Securely: Save access_token in secure storage (Keychain for iOS, EncryptedSharedPreferences for Android)
  2. Store Expiration Date: Save expires_at locally to check token validity offline
  3. Generate Device ID: Use a unique, persistent device identifier (UUID)
  4. Check Token Before Requests: Always verify token expiration before API calls
  5. Handle 401 Responses: Redirect to login screen when receiving 401 errors
  6. Implement Token Refresh: Automatically refresh token when remaining_days â¤ 2 and online
  7. Offline Mode: Cache user data and permissions for offline access

Offline Authentication Flow (Mobile App Logic)

// Pseudocode for mobile app authentication flow

function checkAuthentication() {
    // 1. Check if token exists in local storage
    token = getFromSecureStorage("access_token")
    expiresAt = getFromSecureStorage("token_expires_at")

    if (token == null || expiresAt == null) {
        // No token found - require online login
        return redirectToLogin()
    }

    // 2. Check if token has expired
    if (currentDateTime > expiresAt) {
        // Token expired - require online login
        clearLocalStorage()
        return redirectToLogin()
    }

    // 3. Calculate remaining days
    remainingDays = daysBetween(currentDateTime, expiresAt)

    // 4. If online and token expiring soon, refresh it
    if (isOnline() && remainingDays <= 2) {
        try {
            newToken = refreshToken()
            saveToSecureStorage("access_token", newToken.access_token)
            saveToSecureStorage("token_expires_at", newToken.expires_at)
        } catch (error) {
            // Refresh failed - continue with existing token
            log("Token refresh failed, using existing token")
        }
    }

    // 5. Token valid - allow access
    return allowAccess()
}

function login(email, password) {
    if (!isOnline()) {
        return showError("Internet connection required for first login")
    }

    response = api.post("/api/mobile/auth/login", {
        email: email,
        password: password,
        device_id: getDeviceId(),
        device_name: getDeviceName(),
        device_platform: getPlatform()
    })

    if (response.success) {
        // Save token and user data for offline use
        saveToSecureStorage("access_token", response.data.token.access_token)
        saveToSecureStorage("token_expires_at", response.data.token.expires_at)
        saveToSecureStorage("user_data", response.data.user)

        return redirectToHome()
    } else {
        return showError(response.message)
    }
}

ð± Mobile Sync Strategy
Recommended Sync Flow
  1. Initial Sync: Call GET /api/mobile/categories/sync-all to download all categories
  2. Store Locally: Save all data to SQLite database on device
  3. Save Timestamp: Store the sync_timestamp from response
  4. Incremental Sync: On subsequent syncs, call with ?updated_after={last_sync_timestamp}
  5. Merge Data: Update local database with new/changed records
// Mobile Sync Implementation Example
class CategorySyncService {

    async initialSync() {
        const response = await api.get('/api/mobile/categories/sync-all');

        if (response.success) {
            // Store each category in local database
            for (const [slug, categoryData] of Object.entries(response.data.categories)) {
                await localDB.saveCategory(slug, categoryData.items);
            }

            // Save sync timestamp
            await localDB.setSyncTimestamp('categories', response.data.sync_timestamp);
        }
    }

    async incrementalSync() {
        const lastSync = await localDB.getSyncTimestamp('categories');

        const response = await api.get('/api/mobile/categories/sync-all', {
            params: { updated_after: lastSync }
        });

        if (response.success) {
            // Update only changed records
            for (const [slug, categoryData] of Object.entries(response.data.categories)) {
                for (const item of categoryData.items) {
                    await localDB.upsertCategoryItem(slug, item);
                }
            }

            await localDB.setSyncTimestamp('categories', response.data.sync_timestamp);
        }
    }

    async createItem(categorySlug, name) {
        // Create locally first
        const tempId = generateTempId();
        await localDB.saveItem(categorySlug, { id: tempId, name, pending_sync: true });

        // Sync to server when online
        if (isOnline()) {
            const response = await api.post(`/api/mobile/categories/${categorySlug}`, { name });
            if (response.success) {
                // Update with real ID
                await localDB.updateItemId(categorySlug, tempId, response.data.item.id);
            }
        }
    }
}

ð± Mobile Sync Strategy
Recommended Sync Flow for Central Database
  1. Reserve File IDs: Call POST /file-ids/reserve when online
  2. Store Reservation: Save start_id, end_id, and next_available_id locally
  3. Create Records Offline: Use reserved IDs for new records
  4. Track Used IDs: Increment local next_available_id after each use
  5. Sync When Online: Call POST /data/batch to upload offline records
  6. Report Usage: Call POST /file-ids/sync-used to update server
  7. Request New Batch: When remaining < 1000, request new reservation
// Mobile Implementation Example
class CentralDatabaseSync {

    // Reserve file IDs on app startup when online
    async reserveFileIds() {
        if (!isOnline()) return;

        const currentReservation = await localDB.getActiveReservation();

        // Only request new batch if running low
        if (currentReservation && currentReservation.remaining_count > 1000) {
            return currentReservation;
        }

        const response = await api.post('/api/mobile/database/file-ids/reserve', {
            device_id: getDeviceId(),
            batch_size: 5000
        });

        if (response.success) {
            await localDB.saveReservation(response.data.reservation);
            return response.data.reservation;
        }
    }

    // Get next available file ID for new record
    async getNextFileId() {
        const reservation = await localDB.getActiveReservation();

        if (!reservation || reservation.remaining_count <= 0) {
            throw new Error('No available file IDs. Connect to internet to reserve more.');
        }

        const nextId = reservation.next_available_id;

        // Update local reservation
        await localDB.updateReservation(reservation.id, {
            next_available_id: nextId + 1,
            used_count: reservation.used_count + 1,
            remaining_count: reservation.remaining_count - 1
        });

        return nextId;
    }

    // Create record offline
    async createRecordOffline(recordData) {
        const fileId = await this.getNextFileId();

        const record = {
            ...recordData,
            file_id_number: fileId,
            pending_sync: true,
            created_offline: true,
            created_at: new Date().toISOString()
        };

        await localDB.saveRecord('data', record);
        return record;
    }

    // Sync offline records when online
    async syncOfflineRecords() {
        if (!isOnline()) return;

        const pendingRecords = await localDB.getRecords('data', { pending_sync: true });

        if (pendingRecords.length === 0) return;

        const response = await api.post('/api/mobile/database/data/batch', {
            records: pendingRecords,
            device_id: getDeviceId()
        });

        if (response.success) {
            // Mark synced records
            for (const fileId of [...response.data.created, ...response.data.updated]) {
                await localDB.updateRecord('data', fileId, { pending_sync: false });
            }

            // Sync used count to server
            await this.syncUsedFileIds();
        }
    }

    async syncUsedFileIds() {
        const reservation = await localDB.getActiveReservation();
        if (!reservation) return;

        await api.post('/api/mobile/database/file-ids/sync-used', {
            device_id: getDeviceId(),
            reservation_id: reservation.id,
            used_count: reservation.used_count
        });
    }
}

Future<void> onLoginSuccess(String token) async {
  // Get locally stored codes with their status

  // Call sync endpoint
  final response = await api.post('/mobile/codes/login-sync', {
    'device_id': deviceId,
    'device_codes': localCodes.map((c) => {
      'code': c['code'],
      'used': c['is_used'] == 1,
      'used_at': c['used_at'],
      'record_id': c['record_id']?.toString(),
    }).toList(),
  });

  if (response['success']) {
    final data = response['data'];

    // Delete invalid codes
    for (var invalid in data['sync_result']['invalid_codes']) {
      await localDb.deleteCode(invalid['code']);
    }

    // Save new codes locally
    await localDb.saveNewCodes(data['new_codes']);

    // Update device stats
    await saveDeviceStats(data['device_stats']);
  }
}

// ===================================================
Future<String?> useCodeOffline(String recordType, int recordId) async {
  // Get first unused code
  final code = await localDb.getFirstUnusedCode();
  if (code == null) {
    throw Exception('No codes available! Please connect to internet');
  }

  // Mark code as used locally
  await localDb.markCodeAsUsed(
    code: code,
    usedAt: DateTime.now(),
    recordType: recordType,
    recordId: recordId,
    synced: false,
  );

  return code;
}

// 3. When Coming Online
// ===================================================
Future<void> syncUsedCodes() async {
  // Get unsynced used codes
  final unsyncedCodes = await localDb.getUnsyncedCodes();

  if (unsyncedCodes.isEmpty) return;

  // Send to server
  final response = await api.post('/mobile/codes/confirm-usage', {
    'device_id': deviceId,
    'codes': unsyncedCodes.map((c) => {
      'code': c['code'],
      'used_at': c['used_at'],
      'record_type': c['record_type'],
      'record_id': c['record_id'],
    }).toList(),
  });

  // Update sync status locally
  if (response['success']) {
    for (var code in response['data']['confirmed']) {
      await localDb.markAsSynced(code);
    }
  }
}

// 4. Periodic Balance Check
// ===================================================
Future<void> checkAndRefillCodes() async {
  if (!await isOnline()) return;

  // Get device stats
  final stats = await api.get('/mobile/codes/device-stats', {
    'device_id': deviceId,
  });

  final unusedCount = stats['data']['unused_count'];
  final canRequestMore = stats['data']['can_request_more'];

  // If balance is low, request more
  if (unusedCount < 100 && canRequestMore) {
    final response = await api.post('/mobile/codes/request-codes', {
      'device_id': deviceId,
      'count': 500,
    });

    if (response['success']) {
      await localDb.saveNewCodes(response['data']['codes']);
    }
  }
}

      final total = response.contentLength ?? 0;

      await response.stream.forEach((chunk) {
        sink.add(chunk);
        received += chunk.length;
        onProgress?.call(received, total);
      });

      await sink.close();
      return file;

    } else if (response.statusCode == 403) {
      throw Exception('Unauthorized - Admin role required');
    } else if (response.statusCode == 401) {
      throw Exception('Session expired - Please login again');
    }

    throw Exception('Failed to download file');
  }
}

class PersonsFileDownloader(
    private val baseUrl: String = "https://palestine.benaadev.org",
    private val token: String
) {
    private val client = OkHttpClient()

    /**
     * Download persons file
     * @param destinationFile Target path for saving
     * @param onProgress Progress callback (received, total)
     */
    fun downloadFile(
        destinationFile: File,
        onProgress: ((Long, Long) -> Unit)? = null
    ): Result<File> {
        val request = Request.Builder()
            .url("$baseUrl/api/mobile/database/persons-file/download")
            .addHeader("Authorization", "Bearer $token")
            .build()

        return try {
            val response = client.newCall(request).execute()

            when (response.code) {
                200 -> {
                    val body = response.body ?: throw Exception("Empty response")
                    val contentLength = body.contentLength()

                    FileOutputStream(destinationFile).use { output ->
                        body.byteStream().use { input ->
                            val buffer = ByteArray(8192)
                            var bytesRead: Int
                            var totalRead = 0L

                            while (input.read(buffer).also { bytesRead = it } != -1) {
                                output.write(buffer, 0, bytesRead)
                                totalRead += bytesRead
                                onProgress?.invoke(totalRead, contentLength)
                            }
                        }
                    }
                    Result.success(destinationFile)
                }
                403 -> Result.failure(Exception("Unauthorized - Admin role required"))
                401 -> Result.failure(Exception("Session expired - Login again"))
                404 -> Result.failure(Exception("File not found"))
                else -> Result.failure(Exception("Error: ${response.code}"))
            }
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
}
